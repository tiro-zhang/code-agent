
    neg_sorted = _radix_sort_unsigned(negatives)
    pos_sorted = _radix_sort_unsigned(positives)

    # 负数部分需要倒序并还原符号
    return [-x for x in reversed(neg_sorted)] + pos_sorted

def _radix_sort_unsigned(arr: List[int]) -> List[int]:
    if not arr:
        return []
    max_val = max(arr)
    exp = 1
    output = arr[:]

    while max_val // exp > 0:
        # 对当前位做稳定计数排序（基数为 10）
        count = [0] * 10
        for x in output:
            count[(x // exp) % 10] += 1
        for i in range(1, 10):
            count[i] += count[i - 1]

        temp = [0] * len(output)
        for x in reversed(output):
            digit = (x // exp) % 10
            count[digit] -= 1
            temp[count[digit]] = x
        output = temp
        exp *= 10

    return output
```
